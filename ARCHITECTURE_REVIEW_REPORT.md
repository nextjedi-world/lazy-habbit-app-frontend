### 1. Material 3 Theme Configuration Error

**Location:** `src/_theme.scss` (Lines 156-172)
**Severity:** ❌ BLOCKING
**Impact:** Application cannot build, test, or run

**Issue:**
```scss
// BROKEN: Custom palette format incompatible with mat.define-theme()
$primary-palette: (
  0: #000000,
  10: #001848,
  // ... custom keys like 'primary', 'secondary' are invalid
  primary: #2563EB,  // ❌ NOT ALLOWED
  secondary: #4f7ef7,
  // ...
);

$light-theme: mat.define-theme((
  color: (
    theme-type: light,
    primary: $primary-palette,  // ❌ FAILS HERE
    tertiary: $accent-palette,
  ),
  // ...
));
```

**Error Message:**
```
Expected $config.color.primary to be a valid M3 palette. Got: (0: #000000, 10: #001848, ...)
```

**Root Cause:**
Angular Material's `mat.define-theme()` expects a strict M3 palette generated by `mat.m3-define-palette()`. The custom palette structure with extra keys like `primary`, `secondary`, `tertiary`, `neutral`, etc., is **not valid**.

**Fix Required:**
```scss
// CORRECT APPROACH: Use Material's palette generator
@use '@angular/material' as mat;

$primary-palette: mat.m3-define-palette(mat.$blue-palette, $hue: 60);
$accent-palette: mat.m3-define-palette(mat.$orange-palette, $hue: 50);

// Then define theme
$light-theme: mat.define-theme((
  color: (
    theme-type: light,
    primary: $primary-palette,
    tertiary: $accent-palette,
  ),
  // ...
));
```

**Alternative (Custom Colors):**
If custom colors are needed, define CSS variables separately:
```scss
$light-theme: mat.define-theme((
  color: (
    theme-type: light,
    primary: mat.m3-define-palette(mat.$blue-palette),
    tertiary: mat.m3-define-palette(mat.$orange-palette),
  ),
));

// Apply theme first
@include mat.all-component-themes($light-theme);

// THEN override with CSS variables
:root {
  --color-primary: #2563EB;
  --color-accent: #F97316;
  // ...
}
```

### 6. Constructor Injection in Components

**Location:** Multiple files
**Severity:** ⚠️ HIGH (Pattern inconsistency)
**Impact:** Violates Angular 21 best practices

**Issue:**
Some components use constructor injection instead of `inject()` function:

1. **AuthService** (Line 19):
```typescript
// auth.ts
constructor() {
  this.initializeAuth();
}
```

2. **TaskFormComponent** (Line 63):
```typescript
// task-form.ts
constructor() {
  effect(() => { ... });
  this.taskForm.valueChanges.subscribe(...);
}
```

**Fix Required:**
Move initialization logic out of constructor:

```typescript
// auth.service.ts
export class AuthService {
  private readonly supabase = inject(SupabaseService);
  private readonly router = inject(Router);

  // Use initialization hook instead
  private _currentUser = signal<User | null>(null);
  readonly currentUser = this._currentUser.asReadonly();

  // Initialize in constructor is acceptable for services
  constructor() {
    this.initializeAuth();  // ✅ OK for services
  }
}

// task-form.component.ts
export class TaskFormComponent {
  constructor() {
    // ⚠️ Using constructor for effects is not ideal
    // Consider moving to ngOnInit or using DestroyRef
    effect(() => {
      const taskData = this.task();
      if (taskData) {
        this.taskForm.patchValue(taskData, { emitEvent: false });
      }
    });
  }
}
```

**Better Pattern:**
```typescript
export class TaskFormComponent implements OnInit {
  private destroyRef = inject(DestroyRef);

  ngOnInit() {
    // Subscribe to form changes with automatic cleanup
    this.taskForm.valueChanges
      .pipe(takeUntilDestroyed(this.destroyRef))
      .subscribe(() => {
        if (this.taskForm.valid) {
          this.taskChange.emit(this.taskForm.getRawValue() as TaskFormData);
        }
      });
  }
}
```

### 8. Auth Guard Implementation Issue

**Location:** `src/app/core/guards/auth-guard.ts` (Line 9)
**Severity:** ⚠️ HIGH
**Impact:** Race condition - currentUser signal may be null during initialization

**Issue:**
```typescript
export const authGuard: CanActivateFn = async (route, state) => {
  const authService = inject(AuthService);
  const router = inject(Router);

  const user = authService.currentUser();  // ❌ Signal may not be initialized yet

  if (!user) {
    router.navigate(['/login']);
    return false;
  }

  return true;
};
```

**Problem:**
AuthService constructor calls `initializeAuth()` which is async, but the guard checks the signal synchronously. On first load, `currentUser()` will be null even if the user is logged in.

**Fix Required:**
```typescript
export const authGuard: CanActivateFn = async (route, state) => {
  const authService = inject(AuthService);
  const router = inject(Router);

  // Wait for auth initialization
  const user = await authService.getUser();  // ✅ Add async method

  if (!user) {
    router.navigate(['/login']);
    return false;
  }

  return true;
};

// Add to AuthService:
async getUser(): Promise<User | null> {
  // If already initialized, return immediately
  if (this._currentUser()) {
    return this._currentUser();
  }

  // Otherwise, fetch from Supabase
  return await this.supabase.getCurrentUser();
}
```

### 11. Focus Management on Route Changes

**Location:** `app.ts`
**Severity:** ⚡ MEDIUM
**Impact:** Screen reader users lose context on navigation

**Issue:**
No focus management when navigating between routes.

**Fix Required:**
```typescript
// app.ts
import { Router, NavigationEnd } from '@angular/router';
import { filter } from 'rxjs/operators';

export class App implements OnInit {
  private router = inject(Router);

  ngOnInit() {
    this.router.events
      .pipe(filter(event => event instanceof NavigationEnd))
      .subscribe(() => {
        // Focus main content area
        const mainContent = document.querySelector('main');
        if (mainContent) {
          mainContent.setAttribute('tabindex', '-1');
          (mainContent as HTMLElement).focus();
          // Remove tabindex after focus
          setTimeout(() => mainContent.removeAttribute('tabindex'), 0);
        }
      });
  }
}
```
